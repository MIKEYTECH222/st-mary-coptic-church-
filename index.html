export function initExams(supabase, getContext) { const dashboard = document.getElementById("dashboard"); if (!dashboard || document.getElementById("examsPanel")) return; const nav = dashboard.querySelector(".dashboard-nav"); if (nav) { const link = document.createElement("a"); link.href = "#examsPanel"; link.textContent = "الامتحانات"; nav.insertBefore(link, nav.querySelector("button")); } const panel = document.createElement("section"); panel.id = "examsPanel"; panel.className = "dashboard-block glass"; panel.innerHTML = `

الامتحانات

الامتحانات المتاحة ونتائج محاولاتك.

إنشاء امتحان

إضافة سؤال

نشر أو إلغاء نشر امتحان

`; dashboard.appendChild(panel); const $ = id => document.getElementById(id); const status = text => { $("examStatus").textContent = text || ""; }; let quizzes = []; let groups = []; function isAdmin() { const { currentProfile } = getContext() || {}; return currentProfile?.role === "admin" && currentProfile?.approval_status === "approved"; } function addText(parent, tag, value) { const el = document.createElement(tag); el.textContent = value ?? ""; parent.appendChild(el); return el; } function fillQuizSelect(id) { const select = $(id); select.replaceChildren(); select.add(new Option("اختار امتحانًا", "")); quizzes.forEach(q => select.add(new Option(q.title, q.id))); } async function loadGroups() { const { data, error } = await supabase .from("ministry_groups") .select("id,name") .order("name"); if (error) { status("تعذر تحميل المجموعات: " + error.message); return; } groups = data || []; const select = $("examGroup"); select.replaceChildren(new Option("كل المجموعات", "")); groups.forEach(g => select.add(new Option(g.name, g.id))); } async function loadQuizzes() { const target = $("examList"); target.replaceChildren(); status("جاري تحميل الامتحانات..."); const { data, error } = await supabase .from("quizzes") .select("id,title,description,group_id,published,created_at") .order("created_at", { ascending: false }); if (error) { status("تعذر تحميل الامتحانات: " + error.message); return; } quizzes = data || []; fillQuizSelect("examQuestionQuiz"); fillQuizSelect("examPublishQuiz"); if (isAdmin()) { $("examAdmin").classList.remove("hidden"); } else { $("examAdmin").classList.add("hidden"); } if (!quizzes.length) { addText(target, "p", "لا توجد امتحانات متاحة حاليًا."); status(""); return; } for (const quiz of quizzes) { const card = document.createElement("article"); card.className = "data-item glass"; addText(card, "h3", quiz.title); addText(card, "p", quiz.description || ""); addText(card, "p", quiz.published ? "منشور" : "غير منشور"); if (isAdmin()) { const count = await supabase .from("quiz_questions") .select("id", { count: "exact", head: true }) .eq("quiz_id", quiz.id); addText(card, "p", "عدد الأسئلة: " + (count.count ?? "غير متاح")); } if (quiz.published && !isAdmin()) { const start = document.createElement("button"); start.type = "button"; start.className = "primary"; start.textContent = "بدء الامتحان"; start.onclick = () => startQuiz(quiz); card.appendChild(start); const result = document.createElement("button"); result.type = "button"; result.textContent = "عرض نتيجتي"; result.onclick = () => showResult(quiz); card.appendChild(result); } target.appendChild(card); } status(""); } async function startQuiz(quiz) { const { currentProfile } = getContext() || {}; if (!currentProfile || currentProfile.approval_status !== "approved") { status("لازم يكون حسابك معتمدًا لحل الامتحان."); return; } const { data: questions, error } = await supabase .from("quiz_questions") .select("id,question_text,options,position") .eq("quiz_id", quiz.id) .order("position"); if (error) { status("تعذر تحميل الأسئلة: " + error.message); return; } if (!questions?.length) { status("الامتحان لا يحتوي على أسئلة متاحة."); return; } const target = $("examList"); target.replaceChildren(); addText(target, "h3", quiz.title); const form = document.createElement("form"); form.className = "form-card"; for (const question of questions) { const field = document.createElement("fieldset"); field.style.border = "1px solid var(--border)"; field.style.borderRadius = "12px"; field.style.marginBottom = "16px"; addText(field, "legend", question.question_text); let options = question.options; if (typeof options === "string") { try { options = JSON.parse(options); } catch { options = []; } } if (!Array.isArray(options) || !options.length) { status("صيغة اختيارات أحد الأسئلة غير صحيحة."); return; } options.forEach((option, index) => { const label = document.createElement("label"); const radio = document.createElement("input"); radio.type = "radio"; radio.name = question.id; radio.value = String(index); radio.style.width = "auto"; radio.required = true; label.appendChild(radio); label.appendChild(document.createTextNode(" " + String(option))); field.appendChild(label); }); form.appendChild(field); } const submit = document.createElement("button"); submit.type = "submit"; submit.className = "primary"; submit.textContent = "تسليم الامتحان"; form.appendChild(submit); const cancel = document.createElement("button"); cancel.type = "button"; cancel.textContent = "رجوع"; cancel.onclick = loadQuizzes; form.appendChild(cancel); form.onsubmit = async event => { event.preventDefault(); if (!confirm("متأكد إنك عايز تسلّم الامتحان؟ مش هتقدر تسلّمه مرة ثانية.")) return; submit.disabled = true; status("جاري التصحيح وحفظ النتيجة..."); const answers = {}; questions.forEach(q => { const chosen = form.querySelector(`input[name="${q.id}"]:checked`); if (chosen) answers[q.id] = Number(chosen.value); }); const { data, error } = await supabase.rpc("submit_quiz", { p_quiz_id: quiz.id, p_answers: answers }); submit.disabled = false; if (error) { status("تعذر تسليم الامتحان: " + error.message); return; } const attempt = Array.isArray(data) ? data[0] : data; target.replaceChildren(); addText(target, "h3", "تم تسليم الامتحان"); addText(target, "p", `درجتك: ${attempt?.final_score ?? "—"} من ${attempt?.total_questions ?? "—"}`); status("تم حفظ المحاولة بنجاح."); }; target.appendChild(form); status("جاوب على كل الأسئلة قبل التسليم."); panel.scrollIntoView({ behavior: "smooth", block: "start" }); } async function showResult(quiz) { const { currentUser } = getContext() || {}; if (!currentUser) { status("سجّل الدخول أولًا."); return; } const { data, error } = await supabase .from("quiz_attempts") .select("score,max_score,submitted_at") .eq("quiz_id", quiz.id) .eq("member_id", currentUser.id) .maybeSingle(); if (error) { status("تعذر تحميل النتيجة: " + error.message); return; } if (!data) { status("لم يتم العثور على محاولة مسلّمة لهذا الامتحان."); return; } status(`نتيجتك: ${data.score} من ${data.max_score} — ${new Date(data.submitted_at).toLocaleString("ar-EG")}`); } $("examCreateForm").addEventListener("submit", async event => { event.preventDefault(); if (!isAdmin()) return status("إنشاء الامتحانات متاح للأدمن المعتمد فقط."); status("جاري إنشاء الامتحان..."); const row = { title: $("examTitle").value.trim(), description: $("examDescription").value.trim(), group_id: $("examGroup").value || null, published: false }; const { error } = await supabase.from("quizzes").insert(row); if (error) { status("فشل إنشاء الامتحان: " + error.message); return; } $("examCreateForm").reset(); status("تم إنشاء الامتحان. أضف الأسئلة ثم انشره."); await loadQuizzes(); }); $("examQuestionForm").addEventListener("submit", async event => { event.preventDefault(); if (!isAdmin()) return status("إضافة الأسئلة متاحة للأدمن المعتمد فقط."); const quizId = $("examQuestionQuiz").value; const questionText = $("examQuestionText").value.trim(); const options = [0,1,2,3].map(i => $("examOption" + i).value.trim()); if (!quizId || !questionText || options.some(x => !x)) { status("اختار الامتحان واكتب السؤال والاختيارات الأربعة."); return; } status("جاري حفظ السؤال..."); const { data: existing, error: listError } = await supabase .from("quiz_questions") .select("position") .eq("quiz_id", quizId) .order("position", { ascending: false }) .limit(1); if (listError) { status("تعذر تحديد ترتيب السؤال: " + listError.message); return; } const position = existing?.length ? existing[0].position + 1 : 1; const { data: question, error } = await supabase .from("quiz_questions") .insert({ quiz_id: quizId, question_text: questionText, options, position }) .select("id") .single(); if (error) { status("فشل حفظ السؤال: " + error.message); return; } const { error: keyError } = await supabase .from("quiz_answer_keys") .insert({ question_id: question.id, correct_option: Number($("examCorrectOption").value) }); if (keyError) { status("تم حفظ السؤال لكن تعذر حفظ الإجابة الصحيحة. لا تنشر الامتحان قبل معالجة المشكلة: " + keyError.message); return; } $("examQuestionForm").reset(); status("تم حفظ السؤال والإجابة الصحيحة."); await loadQuizzes(); }); $("examPublishForm").addEventListener("submit", async event => { event.preventDefault(); if (!isAdmin()) return status("النشر متاح للأدمن المعتمد فقط."); const quizId = $("examPublishQuiz").value; const quiz = quizzes.find(q => q.id === quizId); if (!quiz) return status("اختار امتحانًا أولًا."); if (!quiz.published) { const { count, error: countError } = await supabase .from("quiz_questions") .select("id", { count: "exact", head: true }) .eq("quiz_id", quizId); if (countError) return status("تعذر فحص الأسئلة: " + countError.message); if (!count) return status("أضف سؤالًا واحدًا على الأقل قبل النشر."); } const { error } = await supabase .from("quizzes") .update({ published: !quiz.published }) .eq("id", quizId); if (error) { status("تعذر تحديث حالة النشر: " + error.message); return; } status(quiz.published ? "تم إلغاء نشر الامتحان." : "تم نشر الامتحان."); await loadQuizzes(); }); async function refresh() { const { currentProfile } = getContext() || {}; if (!currentProfile || currentProfile.approval_status !== "approved") { $("examAdmin").classList.add("hidden"); status("الامتحانات متاحة للحسابات المعتمدة فقط."); return; } await loadQuizzes(); } const observer = new MutationObserver(() => { if (!dashboard.classList.contains("page-hidden")) refresh(); }); observer.observe(dashboard, { attributes: true, attributeFilter: ["class"] }); loadGroups(); refresh(); }
